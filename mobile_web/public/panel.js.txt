async function api(path, data) {
  const response = await fetch(path, {method: data === undefined ? 'GET' : 'POST', credentials: 'same-origin', cache: 'no-store',
    headers: data === undefined ? {} : {'Content-Type': 'application/json'}, body: data === undefined ? undefined : JSON.stringify(data), signal: AbortSignal.timeout(8000)});
  if (response.status === 401 && path !== '/api/login') { showLogin(); throw new Error('다시 로그인해 주세요.'); }
  if (!response.ok) { let error; try { error = (await response.json()).error; } catch {} throw new Error(error || '연결을 확인해 주세요.'); }
  return response;
}
function showLogin() {
  loggedIn = false;pushChecked=false;testNoticeId=null;closeMenu(); $('login').hidden = false; $('dashboard').hidden = true; $('logout').hidden = true;$('menu-open').hidden=true;

}
function render(data) {
  snapshot = data;
  const status = data.status || {}, online = data.online;
  $('connection').dataset.tone = online ? 'online' : 'offline'; text('connection', online ? 'PC 연결됨' : 'PC 연결 끊김');
  text('updated', '확인 ' + clock(data.updated_at)); text('mode', status.mode || 'COMPYA');
  text('run-state', !online ? 'PC 연결을 기다리는 중' : status.stopping ? '중지하는 중' : status.running ? '매크로 실행 중' : '매크로 대기 중');
  text('status', online ? status.status || '현재 상태를 확인하고 있습니다.' : 'PC 전원·인터넷과 프로그램 실행 여부를 확인하세요.');
  text('season', status.season || '확인 대기'); text('league-count', status.league_count || '확인 대기');
  text('timing', status.timing || '확인 대기'); text('version', status.version ? 'Compya v' + status.version : 'Compya');
  text('blocked', !status.running && online ? status.blocked || (status.dry ? '현재 PC는 모의 실행으로 설정되어 있습니다.' : '') : '');
  const pending = data.command && ['queued', 'delivered'].includes(data.command.state);
  $('start').disabled = busy || pending || !online || !status.can_start || status.running;
  $('stop').disabled = busy || !online || (!status.running && !pending);
  $('reboot').disabled = busy || pending || !online || !status.startup_ready || status.reboot_pending;
  $('cancel-reboot').hidden = !status.reboot_pending;
  $('cancel-reboot').disabled = busy || !online;
  if (status.reboot_pending) { text('run-state', 'PC 재부팅 예약됨'); $('start').disabled = true; }
  if (data.command) { text('command', data.command.message); $('command').dataset.error = ['rejected', 'unknown', 'expired'].includes(data.command.state); }
  if (!online) { $('screen').hidden = true; $('screen-empty').hidden = false; text('screen-empty', '연결이 복구되면 새 게임 화면을 보여 드립니다.'); }
}
async function refresh() {
  if (!loggedIn || document.hidden || refreshing) return;
  refreshing = true;
  try {
    const data = await (await api('/api/status')).json(); render(data);
    if (!pushChecked) {pushChecked=true;checkPush().catch(()=>{pushChecked=false;});}
    if (Date.now()-noticeCheckAt>15000) {noticeCheckAt=Date.now();loadNotices().catch(()=>{});}
    let remember = true; try { remember = localStorage.getItem('compya-remember') !== 'false'; } catch {}
    if (!sessionRenewed && remember) { sessionRenewed = true; api('/api/session/remember', {}).catch(() => { sessionRenewed = false; }); }
    if (data.online && data.screen_at && data.screen_at !== lastScreen) {
      const response = await api('/api/screen'); const url = URL.createObjectURL(await response.blob());
      const previous = imageURL; imageURL = url; $('screen').src = url; $('screen').hidden = false; $('screen-empty').hidden = true;
      lastScreen = data.screen_at; text('screen-time', '촬영 ' + clock(+response.headers.get('X-Captured-At')) + ' · 약 5초마다 갱신');
      if (previous) URL.revokeObjectURL(previous);
    } else if (!data.screen_at) { $('screen').hidden = true; $('screen-empty').hidden = false; text('screen-empty', 'PC에서 새 게임 화면을 기다리고 있습니다.'); }
  } catch (error) {
    if (loggedIn) render({...snapshot, online: false});
    text('command', error.message); $('command').dataset.error = true;
    $('start').disabled = true; $('stop').disabled = true;
  } finally { refreshing = false; }
}

  text('login-note', '연결 확인 중…');
  try {
    await api('/api/login', {key, remember: $('remember').checked}); $('key').value = ''; text('command', ''); loggedIn = true;
    try { localStorage.setItem('compya-remember', String($('remember').checked)); } catch {}
    sessionRenewed = true;
    $('login').hidden = true; $('dashboard').hidden = false; $('logout').hidden = false;$('menu-open').hidden=false; await refresh();
  } catch (error) { text('login-note', error.message); $('login-note').dataset.error = true; }
}
async function send(action, extra = {}) {
  if (busy) return; busy = true; $('start').disabled = true; $('stop').disabled = true; text('command', 'PC에 요청을 보내는 중…');
  try { const result = await (await api('/api/command', {id: crypto.randomUUID(), action, ...extra})).json(); text('command', result.message); }
  catch (error) { text('command', error.message + ' 자동으로 다시 보내지 않습니다.'); $('command').dataset.error = true; }
  finally { busy = false; await refresh(); }
}

function pushSupported() {return 'serviceWorker' in navigator && 'PushManager' in window && 'Notification' in window;}
async function checkPush() {
  if (!pushSupported()) {pushNote('이 브라우저에서 푸시를 지원하지 않습니다. 홈 화면에 추가한 앱에서 다시 확인하세요.');$('push-enable').disabled=true;return;}
  const registration=await navigator.serviceWorker.ready, subscription=await registration.pushManager.getSubscription();
  const registered=subscription && Notification.permission==='granted' && (await (await api('/api/push/check',{endpoint:subscription.endpoint})).json()).registered;
  pushNote(registered?'이 기기에 알림이 연결되어 있습니다. 테스트 알림으로 실제 수신을 확인하세요.':Notification.permission==='denied'?'휴대폰에서 알림이 차단되어 있습니다. 설정에서 허용한 뒤 다시 켜세요.':'메뉴에서 알림을 켜고 알림 권한을 허용하세요.');
  $('push-test').disabled=!registered;$('push-disable').disabled=!subscription;
}
async function loadNotices() {
  const data=await (await api('/api/notifications')).json(), list=$('notice-history');list.replaceChildren();
  text('collection-state',data.collecting?'PC 기록 수집 · '+clock(data.last_received_at)+(data.quality?'':' · 일부 기록 확인 필요'):'PC 기록 수집 연결을 기다리고 있습니다.');
  for (const item of data.items || []) {
    const li=document.createElement('li'),title=document.createElement('strong'),body=document.createElement('p'),time=document.createElement('small');
    title.textContent=item.title;body.textContent=item.body;time.textContent=clock(item.at)+' · '+(item.push_message || ({accepted:'푸시 서버 접수 완료',unconfirmed:'전송 결과 확인 필요','no-device':'휴대폰 알림 연결 전 기록'}[item.push] || '기록됨'));
    if (item.id===testNoticeId) pushNote(item.push_message || '테스트 전송 결과 확인 필요');
    li.append(title,body,time);list.append(li);
  }
  if (!list.children.length) {const li=document.createElement('li');li.textContent='새로운 상태 변화가 생기면 이곳에 기록됩니다.';list.append(li);}
}
$('push-enable').addEventListener('click',async()=>{
  if (!pushSupported()) return;
  $('push-enable').disabled=true;
  try {
    // Permission requests only happen in this explicit button handler.
    const permission=await Notification.requestPermission();
    if (permission!=='granted') throw new Error('알림이 허용되지 않았습니다. 휴대폰 설정에서 이 앱의 알림을 확인하세요.');
    const config=await (await api('/api/push/config')).json();
    if (!config.configured) throw new Error('푸시 연결 준비 중입니다.');
    const registration=await navigator.serviceWorker.ready;
    let subscription=await registration.pushManager.getSubscription();
    if (!subscription) {
      const decoded=atob(config.publicKey.replaceAll('-','+').replaceAll('_','/'));
      subscription=await registration.pushManager.subscribe({userVisibleOnly:true,applicationServerKey:Uint8Array.from(decoded,char=>char.charCodeAt(0))});
    }
    await api('/api/push/subscribe',subscription.toJSON());await checkPush();
  } catch (error) {pushNote(error.message);}
  finally {$('push-enable').disabled=false;}
});

    testNoticeId=null;await checkPush();pushNote('이 기기의 게임 알림을 껐습니다.');
